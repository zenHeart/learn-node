Module

@Module({ 
  controllers: [CatsController],
  providers: [CatsService],
  exports: [CatsService]
})
export class CatsModule {}
@Controller('cats') export class CatsController { constructor( private readonly catsService: CatsService ) {} @Get() findAll() { return this.catsService.findAll(); } @Post() @HttpCode(201) create(@Body() dto: CreateCatDto) { return this.catsService.create(dto); } }
@Injectable() export class CatsService { private cats: Cat[] = []; findAll() { return this.cats; } create(cat: CreateCatDto) { const newCat = { id: Date.now().toString(), ...cat }; this.cats.push(newCat); return newCat; } }
